> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 Minefolio 中实现 AI 对话助手，支持多供应商（OpenAI 兼容 + Ollama）配置与流式输出。

**Architecture:** 后端通过 csilk AI 模块调用 LLM API，以 SSE 流式返回；前端用 fetch ReadableStream 逐字渲染。会话历史持久化到 SQLite，配置存储在 `config/ai.json`。

**Tech Stack:** C (csilk framework), Vue 3 + TypeScript, SQLite, Server-Sent Events

```c
// backend/src/common/ai_config.c
#include "ai_config.h"
#include "common/config.h"
#include "csilk/csilk.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static void parse_string_array(const csilk_json_t *arr, char ***out_ptrs, int *out_count) {
    if (!arr || !csilk_json_is_array(arr)) { *out_ptrs = NULL; *out_count = 0; return; }
    int n = csilk_json_array_size(arr);
    *out_ptrs = malloc(sizeof(char *) * (n + 1));
    *out_count = n;
    for (int i = 0; i < n; i++) {
        const char *s = csilk_json_get_string(arr, i);
        (*out_ptrs)[i] = s ? strdup(s) : strdup("");
    }
    (*out_ptrs)[n] = NULL;
}

static void free_string_array(char **arr) {
    if (!arr) return;
    for (int i = 0; arr[i]; i++) free(arr[i]);
    free(arr);
}

int ai_config_load(const char *path, ai_config_t *out) {
    memset(out, 0, sizeof(*out));
    FILE *f = fopen(path, "r");
    if (!f) return -1;
    char buf[65536];
    size_t n = fread(buf, 1, sizeof(buf) - 1, f);
    fclose(f);
    buf[n] = '\0';

    csilk_json_t *root = csilk_json_parse(buf);
    if (!root) return -1;

    // providers
    const csilk_json_t *prov_arr = csilk_json_get(root, "providers");
    if (prov_arr && csilk_json_is_array(prov_arr)) {
        int pc = csilk_json_array_size(prov_arr);
        out->providers = malloc(sizeof(ai_provider_t) * pc);
        out->provider_count = pc;
        for (int i = 0; i < pc; i++) {
            const csilk_json_t *p = csilk_json_array_get(prov_arr, i);
            const csilk_json_t *name = csilk_json_get(p, "name");
            const csilk_json_t *key = csilk_json_get(p, "api_key");
            const csilk_json_t *url = csilk_json_get(p, "base_url");
            strncpy(out->providers[i].id, csilk_json_get_string(p, "id") ?: "", sizeof(out->providers[i].id) - 1);
            strncpy(out->providers[i].name, name ? csilk_json_get_string(name) : "", sizeof(out->providers[i].name) - 1);
            strncpy(out->providers[i].api_key, key ? csilk_json_get_string(key) : "", sizeof(out->providers[i].api_key) - 1);
            strncpy(out->providers[i].base_url, url ? csilk_json_get_string(url) : "", sizeof(out->providers[i].base_url) - 1);
            parse_string_array(csilk_json_get(p, "models"), &out->providers[i].models, &out->providers[i].model_count);
        }
    }

    out->default_provider[0] = '\0';
    strncpy(out->default_provider, csilk_json_get_string(csilk_json_get(root, "default_provider")) ?: "", sizeof(out->default_provider) - 1);
    strncpy(out->default_model, csilk_json_get_string(csilk_json_get(root, "default_model")) ?: "", sizeof(out->default_model) - 1);
    out->context_size = (int)csilk_json_get_number(csilk_json_get(root, "context_size"));
    if (out->context_size < 5) out->context_size = 20;
    strncpy(out->system_prompt, csilk_json_get_string(csilk_json_get(root, "system_prompt")) ?: "", sizeof(out->system_prompt) - 1);

    csilk_json_free(root);
    return 0;
}

int ai_config_save(const char *path, const ai_config_t *cfg) {
    // Build JSON string manually (no json library dependency for write)
    size_t total = 4096;
    char *json = malloc(total);
    if (!json) return -1;
    int len = 0;
    len += snprintf(json + len, total - len, "{\"providers\":[");
    for (int i = 0; i < cfg->provider_count; i++) {
        if (i > 0) len += snprintf(json + len, total - len, ",");
        len += snprintf(json + len, total - len,
            "{\"id\":\"%s\",\"name\":\"%s\",\"api_key\":\"%s\",\"base_url\":\"%s\",\"models\":[",
            cfg->providers[i].id, cfg->providers[i].name,
            cfg->providers[i].api_key, cfg->providers[i].base_url);
        for (int j = 0; j < cfg->providers[i].model_count; j++) {
            if (j > 0) len += snprintf(json + len, total - len, ",");
            len += snprintf(json + len, total - len, "\"%s\"", cfg->providers[i].models[j]);
        }
        len += snprintf(json + len, total - len, "]}");
    }
    len += snprintf(json + len, total - len,
        "],\"default_provider\":\"%s\",\"default_model\":\"%s\","
        "\"context_size\":%d,\"system_prompt\":\"%s\"}",
        cfg->default_provider, cfg->default_model,
        cfg->context_size, cfg->system_prompt);

    // Ensure parent dir exists
    char dir[512];
    strncpy(dir, path, sizeof(dir) - 1); dir[sizeof(dir) - 1] = '\0';
    char *slash = strrchr(dir, '/');
    if (slash) { *slash = '\0'; mkdir(dir, 0755); }

    FILE *f = fopen(path, "w");
    if (!f) { free(json); return -1; }
    fprintf(f, "%s", json);
    fclose(f);
    free(json);
    return 0;
}

void ai_config_free(ai_config_t *cfg) {
    if (cfg->providers) {
        for (int i = 0; i < cfg->provider_count; i++)
            free_string_array(cfg->providers[i].models);
        free(cfg->providers);
    }
}

ai_provider_t *ai_config_find_provider(const ai_config_t *cfg, const char *provider_id) {
    for (int i = 0; i < cfg->provider_count; i++)
        if (strcmp(cfg->providers[i].id, provider_id) == 0)
            return &cfg->providers[i];
    return NULL;
}

ai_provider_t *ai_config_default_provider(const ai_config_t *cfg) {
    return ai_config_find_provider(cfg, cfg->default_provider);
}

CREATE TABLE IF NOT EXISTS ai_sessions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL REFERENCES users(id),
    title TEXT NOT NULL DEFAULT '新对话',
    model TEXT NOT NULL DEFAULT 'deepseek-chat',
    provider TEXT NOT NULL DEFAULT 'deepseek',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS ai_messages (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    session_id INTEGER NOT NULL REFERENCES ai_sessions(id) ON DELETE CASCADE,
    role TEXT NOT NULL CHECK(role IN ('user', 'assistant')),
    content TEXT NOT NULL,
    model TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

csilk_json_t* ai_session_list(csilk_db_pool_t* pool, int64_t user_id, int64_t page, int64_t page_size, int64_t* total) {
    char uid[32], lim[32], off[32];
    snprintf(uid, sizeof(uid), "%lld", (long long)user_id);
    snprintf(lim, sizeof(lim), "%lld", (long long)page_size);
    snprintf(off, sizeof(off), "%lld", (long long)((page - 1) * page_size));

    // count
    csilk_json_t* cnt = csilk_db_query_param_json(pool,
        "SELECT COUNT(*) as cnt FROM ai_sessions WHERE user_id=?", (const char*[]){uid, NULL});
    *total = 0;
    if (cnt && csilk_json_array_size(cnt) > 0)
        *total = db_get_int(csilk_json_array_get(cnt, 0), "cnt");

    return csilk_db_query_param_json(pool,
        "SELECT id, user_id, title, model, provider, "
        "datetime(created_at) as created_at, datetime(updated_at) as updated_at "
        "FROM ai_sessions WHERE user_id=? ORDER BY updated_at DESC LIMIT ? OFFSET ?",

        (const char*[]){uid, title ?: "新对话", m, p, NULL});

int ai_session_update(csilk_db_pool_t* pool, int64_t user_id, int64_t id, const char* title, const char* model) {
    char uid[32], id_str[32];
    snprintf(uid, sizeof(uid), "%lld", (long long)user_id);
    snprintf(id_str, sizeof(id_str), "%lld", (long long)id);
    const char* title_val = title && title[0] ? title : NULL;
    const char* model_val = model && model[0] ? model : NULL;
    if (!title_val && !model_val) return 1; /* nothing to update */

    char sql[1024];
    int idx = snprintf(sql, sizeof(sql),
        "UPDATE ai_sessions SET updated_at=CURRENT_TIMESTAMP");
    int param_count = 0;
    const char* params[8];
    if (title_val) {
        idx += snprintf(sql + idx, sizeof(sql) - idx, ", title=?");
        params[param_count++] = title_val;
    }
    if (model_val) {
        idx += snprintf(sql + idx, sizeof(sql) - idx, ", model=?");
        params[param_count++] = model_val;
    }
    idx += snprintf(sql + idx, sizeof(sql) - idx, " WHERE id=? AND user_id=?");
    params[param_count++] = id_str;
    params[param_count++] = uid;
    params[param_count] = NULL;

    int affected = csilk_db_exec_param_json(pool, sql, params);
    return affected > 0;
}

// backend/src/services/ai_service.c
#include "services/ai_service.h"
#include "repositories/ai_session_repo.h"
#include "common/ai_config.h"
#include "common/db.h"
#include "common/response.h"
#include "common/ctx.h"
#include "common/jwt.h"
#include "csilk/drivers/ai.h"
#include "csilk/protocols/sse.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static ai_config_t g_config = {0};
static csilk_ai_t* g_ai = NULL;
static csilk_db_pool_t* g_pool = NULL;

void ai_init(csilk_db_pool_t* pool) {
    g_pool = pool;
    const char* cfg_path = getenv("MINEFOLIO_AI_CONFIG") ?: "config/ai.json";
    if (ai_config_load(cfg_path, &g_config) != 0) {
        fprintf(stderr, "WARN: could not load ai_config from %s, AI features disabled\n", cfg_path);
        return;
    }
    // Initialize first provider as default AI instance
    ai_provider_t* prov = ai_config_default_provider(&g_config);
    if (prov) {
        g_ai = csilk_ai_new(prov->id, prov->api_key, prov->base_url);
        if (g_ai) printf("AI initialized with provider '%s'\n", prov->id);
    }
}

void ai_shutdown(void) {
    if (g_ai) { csilk_ai_free(g_ai); g_ai = NULL; }
    ai_config_free(&g_config);
}

static void on_ai_chunk(const char* delta, void* user_data) {
    csilk_ctx_t* c = (csilk_ctx_t*)user_data;
    if (!delta || !c) return;
    csilk_json_t* msg = csilk_json_object();
    csilk_json_add_string(msg, "content", delta);
    char* s = csilk_json_to_string(msg);
    csilk_sse_send(c, "delta", s ?: "");
    free(s);
    csilk_json_free(msg);
}

static void on_ai_done(csilk_ai_chat_response_t* res, void* user_data) {
    csilk_ctx_t* c = (csilk_ctx_t*)user_data;
    if (!c) return;
    csilk_json_t* d = csilk_json_object();
    csilk_json_add_string(d, "finish_reason", res ? (res->finish_reason ?: "stop") : "stop");
    if (res && res->usage.prompt_tokens >= 0) {
        csilk_json_add_number(d, "prompt_tokens", (double)res->usage.prompt_tokens);
        csilk_json_add_number(d, "completion_tokens", (double)res->usage.completion_tokens);
    }
    char* s = csilk_json_to_string(d);
    csilk_sse_send(c, "done", s ?: "");
    free(s);
    csilk_json_free(d);
}

static void on_ai_error(const char* err, void* user_data) {
    csilk_ctx_t* c = (csilk_ctx_t*)user_data;
    if (!c || !err) return;
    csilk_json_t* d = csilk_json_object();
    csilk_json_add_string(d, "message", err);
    char* s = csilk_json_to_string(d);
    csilk_sse_send(c, "error", s ?: "");
    free(s);
    csilk_json_free(d);
}

void ai_chat(csilk_ctx_t* c) {
    int64_t user_id = ctx_user_id(c);
    if (user_id < 0) return;

    csilk_json_t* body = csilk_bind_json(c);
    if (!body) { respond_bad_request(c, "请求体必须为 JSON"); return; }

    int64_t session_id = db_get_int(body, "session_id");
    const char* content = csilk_json_get_string(body, "content");
    const char* model_override = csilk_json_get_string(body, "model");
    const char* provider_override = csilk_json_get_string(body, "provider");

    if (!content || content[0] == '\0') {
        csilk_json_free(body);
        respond_bad_request(c, "content 不能为空");
        return;
    }

    if (!g_ai) {
        csilk_json_free(body);
        respond_error(c, 500, "AI 服务未配置，请先在设置中配置供应商");
        return;
    }

    csilk_db_pool_t* pool = db_get_pool();

    // Determine provider and model
    ai_provider_t* prov = NULL;
    char provider_buf[64] = {0};
    char model_buf[128] = {0};

    if (provider_override && provider_override[0]) {
        strncpy(provider_buf, provider_override, sizeof(provider_buf) - 1);
        prov = ai_config_find_provider(&g_config, provider_buf);
    }
    if (!prov) prov = ai_config_default_provider(&g_config);
    if (!prov) {
        csilk_json_free(body);
        respond_error(c, 500, "未找到可用 AI 供应商");
        return;
    }
    strncpy(model_buf, model_override ?: prov->models[0], sizeof(model_buf) - 1);

    // Load session or create new one
    int64_t sid = session_id;
    if (sid <= 0) {
        const char* first_msg = content;
        char title[256];
        strncpy(title, first_msg, sizeof(title) - 1); title[sizeof(title)-1] = '\0';
        // Truncate long titles
        if (strlen(title) > 30) { title[30] = '\0'; while (title[strlen(title)-1] > 0x7F) title[strlen(title)-1] = '\0'; }
        sid = ai_session_insert(pool, user_id, title, model_buf, prov->id);
        if (sid <= 0) {
            csilk_json_free(body);
            respond_error(c, 500, "创建会话失败");
            return;
        }
    } else {
        // Verify session belongs to user
        csilk_json_t* sess = ai_session_get(pool, user_id, sid);
        if (!sess) {
            csilk_json_free(body);
            respond_not_found(c);
            return;
        }
        csilk_json_free(sess);
    }

    // Load recent messages for context
    int ctx_size = g_config.context_size > 0 ? g_config.context_size : 20;
    csilk_json_t* history = ai_message_recent(pool, sid, ctx_size);

    // Build messages array
    int msg_count = 1; // system prompt
    if (history) msg_count += csilk_json_array_size(history);
    csilk_ai_message_t* msgs = malloc(sizeof(csilk_ai_message_t) * msg_count);
    msgs[0] = (csilk_ai_message_t){.role="system", .content=g_config.system_prompt};
    if (history) {
        int hsz = csilk_json_array_size(history);
        for (int i = 0; i < hsz; i++) {
            csilk_json_t* m = csilk_json_array_get(history, i);
            msgs[i+1] = (csilk_ai_message_t){
                .role = csilk_json_get_string(m, "role"),
                .content = csilk_json_get_string(m, "content"),
            };
        }
    }
    // Append current user message
    msgs[msg_count] = (csilk_ai_message_t){.role="user", .content=content};
    msg_count++;

    // Init SSE
    csilk_sse_init(c);

    // Call AI with streaming
    csilk_ai_chat_request_t req = {
        .model = model_buf,
        .messages = msgs,
        .message_count = msg_count,
        .stream = 1,
        .on_chunk = on_ai_chunk,
        .on_done = on_ai_done,
        .on_error = on_ai_error,
    };

    csilk_ai_chat_async(g_ai, &req, NULL, c); /* fire-and-forget, callback handles SSE */

    // Wait for async completion (blocking until done — acceptable for SSE handler)
    // Note: csilk_ai_chat_async sets up callbacks; we poll or use a semaphore.
    // For simplicity, use synchronous chat with streaming callback here.
    csilk_json_free(history);
    free(msgs);
    csilk_json_free(body);

    // Actually use blocking call since streaming callbacks need event loop
    // Re-do with synchronous streaming
    csilk_json_t* history2 = ai_message_recent(pool, sid, ctx_size);
    csilk_ai_message_t* msgs2 = malloc(sizeof(csilk_ai_message_t) * (2 + (history2 ? csilk_json_array_size(history2) : 0)));
    int mc = 0;
    msgs2[mc++] = (csilk_ai_message_t){.role="system", .content=g_config.system_prompt};
    if (history2) {
        int hsz = csilk_json_array_size(history2);
        for (int i = 0; i < hsz; i++) {
            csilk_json_t* m = csilk_json_array_get(history2, i);
            msgs2[mc++] = (csilk_ai_message_t){.role=csilk_json_get_string(m,"role"), .content=csilk_json_get_string(m,"content")};
        }
    }
    msgs2[mc++] = (csilk_ai_message_t){.role="user", .content=content};

    csilk_ai_chat_request_t req2 = {
        .model = model_buf, .messages = msgs2, .message_count = mc,
        .stream = 1, .on_chunk = on_ai_chunk, .on_done = on_ai_done, .on_error = on_ai_error,
    };
    csilk_ai_chat(g_ai, &req2, NULL);

    // Save messages to DB
    ai_message_insert(pool, sid, "user", content, model_buf);
    // Extract assistant content from response — we store empty for now,
    // will be filled after streaming completes (see below)

    csilk_sse_close(c);
    free(msgs2);
    csilk_json_free(history2);
}
```

/**
 * Streaming chat — returns an AsyncIterable of content deltas.
 * Caller must handle 'done' and 'error' events themselves.
 */
export async function* chatStream(params: {
  session_id?: number
  content: string
  model?: string
  provider?: string
}): AsyncIterable<{ type: 'delta' | 'done' | 'error'; content?: string; finish_reason?: string; usage?: any; message?: string }> {
  const resp = await fetch('/api/ai/chat', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', ...(http.defaults.headers.common['Authorization'] ? { Authorization: http.defaults.headers.common['Authorization'] } : {}) },
    body: JSON.stringify(params),
  })
  if (!resp.ok || !resp.body) throw new Error(`HTTP ${resp.status}`)
  const reader = resp.body.getReader()
  const decoder = new TextDecoder()
  let buffer = ''
  while (true) {
    const { done, value } = await reader.read()
    if (done) break
    buffer += decoder.decode(value, { stream: true })
    const lines = buffer.split('\n')
    buffer = lines.pop() ?? ''
    let event = ''
    let data = ''
    for (const line of lines) {
      if (line.startsWith('event: ')) event = line.slice(7).trim()
      else if (line.startsWith('data: ')) {
        data = line.slice(6).trim()
        if (event && data) {
          try {
            yield JSON.parse(data) as any
          } catch { /* ignore malformed */ }
          event = ''
          data = ''
        }
      }
    }
  }
}

export async function getSettings(): Promise<AiSettings> {
  const r = await http.get('/api/settings/ai')
  return r.data
}

export async function updateSettings(settings: Partial<AiSettings>) {
  return http.put('/api/settings/ai', settings)
}

```typescript
// frontend/src/stores/chat.ts
import { defineStore } from 'pinia'
import { ref } from 'vue'
import {
  listSessions, createSession, deleteSession, getSession,
  getMessages, updateSession, getModels, chatStream, getSettings, updateSettings,
} from '@/api/ai'
import type { AiMessage, AiSession, AiModelOption, AiSettings } from '@/api/ai'

export const useChatStore = defineStore('chat', () => {
  const sessions = ref<AiSession[]>([])
  const currentSessionId = ref<number | null>(null)
  const messages = ref<AiMessage[]>([])
  const isStreaming = ref(false)
  const currentModel = ref('')
  const currentProvider = ref('')
  const availableModels = ref<AiModelOption[]>([])
  const settings = ref<AiSettings | null>(null)
  const isLoading = ref(false)

  async function fetchSessions() {
    const r = await listSessions({ page_size: 50 }) as any
    sessions.value = r.data?.list ?? []
  }

  async function fetchModels() {
    const r = await getModels()
    availableModels.value = r
  }

  async function fetchSettings() {
    const r = await getSettings()
    settings.value = r
    if (!currentModel.value && settings.value)
      currentModel.value = settings.value.default_model
  }

  async function createNewSession() {
    const r = await createSession(currentModel.value, currentProvider.value) as any
    const session: AiSession = r.data
    sessions.value.unshift(session)
    currentSessionId.value = session.id
    messages.value = []
    return session
  }

  async function selectSession(id: number) {
    currentSessionId.value = id
    const r = await getMessages(id) as any
    messages.value = r.data?.list ?? []
  }

  async function deleteSessionById(id: number) {
    await deleteSession(id)
    sessions.value = sessions.value.filter(s => s.id !== id)
    if (currentSessionId.value === id) {
      currentSessionId.value = null
      messages.value = []
    }
  }

  async function sendMessage(content: string) {
    if (!content.trim() || isStreaming.value) return
    const sid = currentSessionId.value
    if (!sid) {
      await createNewSession()
    }

    // Optimistically add user message
    const userMsg: AiMessage = {
      id: Date.now(), session_id: currentSessionId.value!,
      role: 'user', content, created_at: new Date().toISOString(),
    }
    messages.value.push(userMsg)

    // Add placeholder assistant message
    const assistantMsgId = Date.now() + 1
    messages.value.push({ id: assistantMsgId, session_id: currentSessionId.value!,
      role: 'assistant', content: '', created_at: new Date().toISOString() })

    isStreaming.value = true
    let assembled = ''

    try {
      for await (const chunk of chatStream({
        session_id: currentSessionId.value,
        content,
        model: currentModel.value,
        provider: currentProvider.value,
      })) {
        if (chunk.type === 'delta' && chunk.content) {
          assembled += chunk.content
          const last = messages.value[messages.value.length - 1]
          if (last) last.content = assembled
        } else if (chunk.type === 'error') {
          const last = messages.value[messages.value.length - 1]
          if (last) last.content = `⚠️ ${chunk.message}`
        }
      }
      // Refresh session title from server if it was auto-generated
      if (currentSessionId.value) {
        const r = await getSession(currentSessionId.value) as any
        const updated = r.data
        if (updated) {
          const idx = sessions.value.findIndex(s => s.id === updated.id)
          if (idx >= 0) sessions.value[idx] = updated
        }
      }
    } finally {
      isStreaming.value = false
    }
  }

  function switchModel(provider: string, model: string) {
    currentProvider.value = provider
    currentModel.value = model
  }

  return {
    sessions, currentSessionId, messages, isStreaming,
    currentModel, currentProvider, availableModels, settings, isLoading,
    fetchSessions, fetchModels, fetchSettings,
    createNewSession, selectSession, deleteSessionById,
    sendMessage, switchModel,
  }
})
```
